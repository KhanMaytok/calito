#############################################
# CALITO - Evento del scenario "5 Arks" en partida
#
# Este es el evento "de verdad": el que veria el jugador. Los eventos
# de 00_calito_test_events.txt son solo para depuracion.
#
# calito.100  -> Ventana inicial: el mundo se acabo, elegimos que clase
#                de arco somos. Lanza el scenario completo.
#                tanda de asari se une al imperio.
#############################################

namespace = calito

# --- Ventana inicial -------------------------------------------------------
country_event = {
	id = calito.100
	title = "calito.100.name"
	desc = "calito.100.desc"
	picture = GFX_evt_star_white
	show_sound = event_super_explosion
	location = root.capital_scope

	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		calito_five_arks_available = yes
		calito_has_nomads = yes
		exists = owner
	}

	immediate = {
		set_country_flag = calito_five_arks_pending
	}

	# --- Opcion A: arco civil (el mas grande, mas poblacion) ---
	option = {
		name = calito.100.a
		custom_tooltip = calito.100.a.tt
		hidden_effect = {
			calito_five_arks_setup_effect = {
				ARK_TYPE = civilian_arkship
			}
			country_event = { id = calito.110 days = 5 }
		}
		ai_chance = { factor = 3 }
	}

	# --- Opcion B: arco militar (menos gente, mas fueronza) ---
	option = {
		name = calito.100.b
		custom_tooltip = calito.100.b.tt
		hidden_effect = {
			calito_five_arks_setup_effect = {
				ARK_TYPE = military_arkship
			}
			country_event = { id = calito.110 days = 5 }
		}
		ai_chance = { factor = 1 }
	}

	# --- Opcion C: arco cientifico (el mas pequeno, el mas listo) ---
	option = {
		name = calito.100.c
		custom_tooltip = calito.100.c.tt
		hidden_effect = {
			calito_five_arks_setup_effect = {
				ARK_TYPE = science_arkship
			}
			country_event = { id = calito.110 days = 5 }
		}
		ai_chance = { factor = 2 }
	}

	# --- Opcion D: rechazar (el mundo sigue, tu no cambias) ---
	option = {
		name = calito.100.d
		default_hide_option = yes
		custom_tooltip = calito.100.d.tt
		hidden_effect = {
			remove_country_flag = calito_five_arks_pending
			set_country_flag = calito_five_arks_aborted
		}
	}
}

# --- Confirmacion ----------------------------------------------------------
country_event = {
	id = calito.110
	title = "calito.110.name"
	desc = "calito.110.desc"
	picture = GFX_evt_star_white
	show_sound = event_celebration
	location = root.capital_scope

	is_triggered_only = yes

	option = {
		name = calito.110.a
		custom_tooltip = calito.110.a.tt
		hidden_effect = {
			remove_country_flag = calito_five_arks_pending
			# Los asari YA vienen a bordo desde el setup: son la
			# poblacion de los cinco arcos. No hay segunda tanda que
			# ofrecer, asi que se marca directamente como resuelto.
			set_country_flag = calito_five_arks_asari_done
		}
	}
}
