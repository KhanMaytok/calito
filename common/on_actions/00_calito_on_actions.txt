# eventos de consola de 00_calito_test_events.txt funcionan igual.

# --- Paso 1: en el inicio de partida, comprobar si toca el scenario ---
# Un gatekeeper con dias de retardo: deja que la partida se asiente
# antes de soltarle la ventana dramatica encima al jugador.
on_game_start_country = {
	events = {
		calito.gatekeeper
	}
}

# --- Paso 2: cada ano ---
#   - Si el jugador nunca eligio en calito.100, se re-ofrece.
#   - Si ya esta en marcha, se le ofrece la tanda de asari.
#   Solo se dispara uno de los dos: cada uno exige el otro estado.
on_yearly_pulse_country = {
	events = {
		calito.asari_gate
		calito.asari_yearly
	}
}
