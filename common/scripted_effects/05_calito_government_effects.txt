#############################################
# CALITO - Gobierno: monarquia absolutista
#
# !!! LA AUTORIDAD IMPERIAL ES 'auth_imperial', NO 'auth_oligarchic' !!!
# Este fichero antes usaba auth_oligarchic, que es un error. El trigger
# de vanilla es literally:
#     common/scripted_triggers/00_scripted_triggers.txt:1822
#         is_imperial_authority = { has_authority = auth_imperial }
# Es decir: SOLO auth_imperial cuenta como imperial. Las cuatro
# autoridades base se diferencian asi:
#
#     autoridad            has_heir   ruler_council_position
#     auth_democratic         no         councilor_ruler_democratic
#     auth_oligarchic      no         councilor_ruler_oligarchic
#     auth_dictatorial     no         councilor_ruler_dictatorial
#     auth_imperial        SI         councilor_ruler_imperial
#
# Consecuencias de usar la equivocada (auth_oligarchic):
#   - has_heir = no  ->  SIN DINASTIA, sin herederos, sin Succession
#   - is_imperial_authority = no  ->  NINGUN gobierno imperial era posible
#     (gov_star_empire, gov_despotic_empire, gov_despotic_hegemony,
#      gov_feudal_empire, gov_martial_empire, gov_theocratic_monarchy...)
#     y el imperio caia en gov_military_junta o gov_executive_committee
#   - el trono era councilor_ruler_oligarchic, no councilor_ruler_imperial
#
# ETICAS QUE BLOQUEAN auth_imperial (00_authorities.txt, bloque possible):
#     ethic_gestalt_consciousness, ethic_egalitarian,
#     ethic_fanatic_egalitarian
# OJO: el fanatico EGUALITARIO no se puede quitar con script, asi que si
# el jugador lo tiene, esta autoridad no es posible y change_government
# falla en silencio. Se avisa con log.
#
# !!! EL ORDEN IMPORTA !!!
# change_government RECALCULA el gobierno con las ETICAS que hay en ese
# momento. Antes se hacia change_government PRIMERO y las eticas
# DESPUES, asi que no caia en gov_star_empire. Ahora: eticas primero.
#
# QUE GOBIERNO SALE: con auth_imperial + materialista + militarista, los
# candidatos son (de 170 gobiernos de vanilla, estos son los que aceptan
# is_imperial_authority = yes sin exigir un pais especial):
#     gov_star_empire        RT_EMPEROR / RT_EMPRESS  (militarista)
#     gov_despotic_hegemony  RT_OVERLORD               (materialista)
#     gov_despotic_empire    RT_EMPEROR                (sin requisito)
# Usamos el civico aristocratic_elite, que no toca eticas, para no
# Fight el gobierno.
#
# ALTERNATIVAS de sabor, cambiando solo el gobierno:
#   + ethic_spiritualist -> gov_theocratic_monarchy  (RT_ARCHPROPHET)
#   + ethic_pacifist     -> gov_irenic_monarchy      (RT_HIGH_KING)
#   + civic_technocracy  -> gov_illuminated_autocracy
#############################################

calito_setup_absolute_monarchy_effect = {
	# --- 1) ETICAS PRIMERO ---
	# change_government las lee en cuanto se ejecuta.
	if = {
		limit = {
			OR = {
				is_fanatic_egalitarian = yes
				has_ethic = ethic_gestalt_consciousness
			}
		}
		log_error = "calito: etico FANATICO IGUALITARIO o conciencia gestalt. auth_imperial NO es posible con ellos, asi que el imperio no tendra dinastia. Cambia la etica a mano y vuelve a ejecutar."
	}

	# Eticos no fanaticos: se pueden quitar sin problema.
	if = {
		limit = { is_egalitarian = yes }
		country_remove_ethic = ethic_egalitarian
	}
	if = {
		limit = { is_spiritualist = yes }
		country_remove_ethic = ethic_spiritualist
	}
	if = {
		limit = { is_pacifist = yes }
		country_remove_ethic = ethic_pacifist
	}
	if = {
		limit = { is_xenophile = yes }
		country_remove_ethic = ethic_xenophile
	}
	if = {
		limit = { NOT = { has_ethic = ethic_materialist } }
		country_add_ethic = ethic_materialist
	}
	if = {
		limit = { NOT = { has_ethic = ethic_militarist } }
		country_add_ethic = ethic_militarist
	}

	# --- 2) Autoridad imperial: da herederos (has_heir = yes) y el cargo
	#        del trono, councilor_ruler_imperial. ---
	change_government = {
		authority = auth_imperial
		civics = random
		remove_invalid_civics = yes
	}

	# --- 3) Verificacion: change_government falla en silencio si la
	#        autoridad no es posible, asi que lo comprobamos. ---
	if = {
		limit = {
			NOT = { has_authority = auth_imperial }
		}
		log_error = "calito: change_government NO pudo aplicar auth_imperial. Causa habitual: etico FANATICO IGUALITARIO (no se puede quitar por script) o conciencia gestalt."
	}

	# --- 4) Titulos propios: el trono no se llama "Emperatriz" de serie. ---
	set_ruler_title_female = "RT_CALITO_QUEEN"
	set_ruler_title_male = "RT_CALITO_KING"

	# --- 5) Civic aristocratic_elite: habilita arkship_noble_estates, que
	#        es la Mansion Noble del arco. Thematico para una monarquia. ---
	if = {
		limit = { NOT = { has_civic = civic_aristocratic_elite } }
		force_add_civic = civic_aristocratic_elite
	}

	# --- 6) Registro para poder verificarlo despues. ---
	set_country_flag = calito_absolute_monarchy
}
