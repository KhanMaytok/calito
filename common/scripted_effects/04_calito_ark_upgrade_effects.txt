#############################################
# CALITO - Mejora de arcos a late game
#
# Sube un arco de tier 1 a tier 3 y lo llena de modulos y edificios,
# como si el imperio llevase decades explorando.
#
# POR QUE NO USAMOS EL EFECTO VANILLA create_arkship_effect:
#   Solo crea el arco. No lo sube de tier, no le pone modulos.
#
# POR QUE set_starbase_size NO ES TRAMPA:
#   El efecto no comprueba nada. Se puede llamar en cualquier momento
#   con cualquier tamano. Pero los tier 2 y 3 tienen
#   potential_construction = { always = no }, asi que no se pueden
#   CONSTRUIR: solo existen por set_starbase_size. Aun asi damos las
#   technologies antes, para que la UI y elDiseñador resuelvan bien.
#
# RANURAS (common/starbase_levels/06_arkship_levels.txt):
#   tier 1: modulos 1 2        | edificios 1 2 3 4
#   tier 2: modulos 1 2 3 4    | edificios 1..8
#   tier 3: modulos 1..6 + spinal_mount_1 | edificios 1..8 + lower_1..4
#
# OJO: el slot 7 de modulos ES el spinal_mount_1. El indice 7 no es
# "el septimo modulo", es un ranur de tipo especial.
#
# ARMA: cada modulo de arma trae un component_set, y el motor lo
# resuelve automaticamente al nivel mas alto que hayamos investigado.
# Por eso damos technologies de armas: no hace falta (ni se puede)
# elegir el arma concreta.
#############################################

# Sube UN arco a tier 3 y lo kitea.
# SCOPE: colonia (pc_ark) o flota. Se llama desde la colonia del arco.
# $ARK_TYPE$ = civilian_arkship | science_arkship | military_arkship
calito_max_out_ark_effect = {
	starbase = {   # fleet/starbase se eliminaron: el error 'Invalid context switch [fleet] from [country]' venia de que este efecto se llamaba en scope de PAIS. starbase = { } es lo que usa vanilla en initialize_arkship_starbase_effect (nomads_effects.txt:3405), invocado en scope de NAVE.
		# --- Tamaño ---
		set_starbase_size = $ARK_TYPE$_tier_3

		# --- Modulos, ranuras 1-6 ---
		# 1: ingenieria (reparacion + construccion)
		set_starbase_module = { slot = 1 module = arkship_engineering_bay }
		# 2: muelle (capacidad naval)
		set_starbase_module = { slot = 2 module = arkship_docking_bay }
		# 3: Bahia de lanceros (naves de ataque)
		set_starbase_module = { slot = 3 module = arkship_hangar_bay }
		# 4: bateria de energia
		set_starbase_module = { slot = 4 module = arkship_lance_battery }
		# 5: defensa cinetica pesada
		set_starbase_module = { slot = 5 module = arkship_mass_accelerator }
		# 6: matraque (las de verdad)
		set_starbase_module = { slot = 6 module = arkship_missile_battery }

		# --- Modulo especial, ranura 7 = spinal_mount_1 ---
		# El devastador es un ARMA titanic. Lo ponemos solo en arcos
		# militares; en los demas, el arsenal (permite construir).
		set_starbase_module = { slot = 7 module = arkship_capital_shipyard }

		# --- Edificios, ranuras 1-12 ---
		# 1-4: armadura y escudo (cada edificio = 3 componentes)
		set_starbase_building = { slot = 1  building = arkship_archaeo_armor }
		set_starbase_building = { slot = 2  building = arkship_archaeo_armor }
		set_starbase_building = { slot = 3  building = arkship_archaeo_shield }
		set_starbase_building = { slot = 4  building = arkship_archaeo_shield }
		# 5-10: la economia de una colonia tardia
		set_starbase_building = { slot = 5  building = arkship_energy_hub }
		set_starbase_building = { slot = 6  building = arkship_food_hub }
		set_starbase_building = { slot = 7  building = arkship_alloys_hub }
		set_starbase_building = { slot = 8  building = arkship_consumer_goods_hub }
		set_starbase_building = { slot = 9  building = arkship_bureaucracy_hub }
		set_starbase_building = { slot = 10 building = arkship_trade_hub }
		# 11-12: sin limite de instancias, para crecimiento y recurso
		set_starbase_building = { slot = 11 building = arkship_hydroponics_bay }
		set_starbase_building = { slot = 12 building = arkship_harvest_processing_arrays }
	}
}
